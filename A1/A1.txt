             (clear ?obj - object )
             (is_table ?obj - object)

(:action MOVE
:parameters ( ?b - block ?obj1 - object ?obj2 - object) 
:precondition (and (on ?b ?obj1) (clear ?b) (clear ?obj2))
:effect (and 
  (on ?b ?obj2) 
  (not (on ?b ?obj1)) 
  (clear ?obj1) 
  (when (not (is_table ?obj2)) (not (clear ?obj2)))
