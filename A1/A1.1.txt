(define (domain blocks_world ) 
(:requirements :strips :typing) 
(:types block table - object
) 
(:predicates (on ?obj1 - block ?obj2 - object)
             (clear ?obj - object )
             (is_table ?obj - object)
) 
(:action MOVE
:parameters ( ?b - block ?obj1 - object ?obj2 - object) 
:precondition (and (on ?b ?obj1) (clear ?b) (clear ?obj2))
:effect (and 
  (on ?b ?obj2) 
  (not (on ?b ?obj1)) 
  (clear ?obj1) 
  (when (not (is_table ?obj2)) (not (clear ?obj2)))
)
)
) 

(define (problem swap_blocks) 
 (:domain blocks_world) 
(:objects A B  - block
 T - table
 ) 
(:init (on A B)(on B T)(clear A)(clear T)(is_table T)) 
(:goal (and (on B A)(on A T))) 
)

a.
Assign(a1, a2, val1, val2)
Precondition:   
    Variable(a1) ^ Variable(a2) ^ Value(val1) ^ Value(val2) 
    ^ HasValue(a1, val1) ^ HasValue(a2, val2)
Effect: HasValue(a2, val1) ^ not HasValue(a2, val2)

b.
Assume we have a value null.

Have another action schema for object creation
Create(a)
Precondition: Address(a) ^ not Variable(a)
Effect: Variable(a) ^ HasValue(a, null)

Initial state:
Variable(a1) ^ Variable(a2) ^ Value(val1) ^ Value(val2) 
^ HasValue(a1, val1) ^ HasValue(a2, val2) 
^ Address(a3) ^ Value(null)

Goal state:
HasValue(a1, val2) ^ HasValue(a2, val1)

Then the following plan is valid for the task
Create(a3)
Assign(a1, a3, val1, null)
Assign(a2, a1, val2, val1)
Assign(a3, a2, val1, val2)
